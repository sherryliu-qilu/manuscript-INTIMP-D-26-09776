# ---- 1. Read 10x-like mtx / genes / barcodes, one sample at a time ----
read_one_sample <- function(gsm, patient, figo, stage) {
  prefix <- file.path(base_dir, paste0(gsm, "_", patient))
  mtx_path <- paste0(prefix, ".matrix.mtx.gz")
  gene_path <- paste0(prefix, ".genes.tsv.gz")
  barcode_path <- paste0(prefix, ".barcodes.tsv.gz")
  if (!all(file.exists(c(mtx_path, gene_path, barcode_path)))) {
    stop("Missing input file(s) for ", patient, ": ", prefix)
  }
  counts <- Matrix::readMM(mtx_path)
  genes <- read.delim(gzfile(gene_path), header = FALSE, sep = "\t", quote = "", check.names = FALSE)
  barcodes <- read.delim(gzfile(barcode_path), header = FALSE, sep = "\t", quote = "", check.names = FALSE)
  if (nrow(counts) != nrow(genes) || ncol(counts) != nrow(barcodes)) {
    stop("Dimension mismatch in ", patient)
  }
  # In 10x gene files, column 2 is generally the gene symbol.
  gene_names <- if (ncol(genes) >= 2) as.character(genes[[2]]) else as.character(genes[[1]])
  rownames(counts) <- make.unique(gene_names)
  colnames(counts) <- paste0(patient, "_", barcodes[[1]])
  counts <- as(counts, "dgCMatrix")
  obj <- CreateSeuratObject(counts = counts, project = patient, min.cells = 0, min.features = 0)
  obj$Patient <- patient
  obj$GSM <- gsm
  obj$FIGO <- figo
  obj$Stage <- stage
  obj[["percent.mt"]] <- PercentageFeatureSet(obj, pattern = "^MT-")
  obj
}

seurat_list <- lapply(seq_len(nrow(sample_info)), function(i) {
  x <- sample_info[i, ]

# ---- 2. QC (previously used thresholds) ----
qc_before <- bind_rows(lapply(names(seurat_list), function(s) {

# ---- 3. scDblFinder: ACTUAL doublet detection per sample ----
# Run independently per library/sample. The expected doublet rate is estimated
# by scDblFinder; do not substitute invented removal counts.
doublet_results <- vector("list", length(seurat_qc))
names(doublet_results) <- names(seurat_qc)
seurat_singlets <- vector("list", length(seurat_qc))
names(seurat_singlets) <- names(seurat_qc)
for (s in names(seurat_qc)) {
  message("Running scDblFinder for ", s)
  obj <- seurat_qc[[s]]
  sce <- as.SingleCellExperiment(obj, assay = "RNA")
  sce <- scDblFinder::scDblFinder(sce, verbose = FALSE)
  obj$scDblFinder.score <- as.numeric(colData(sce)$scDblFinder.score)
  obj$scDblFinder.class <- as.character(colData(sce)$scDblFinder.class)
  doublet_results[[s]] <- data.frame(
    Patient = s,
    Cell_Barcode = colnames(obj),
    scDblFinder_score = obj$scDblFinder.score,
    scDblFinder_class = obj$scDblFinder.class,
    stringsAsFactors = FALSE
  )
  seurat_singlets[[s]] <- subset(obj, subset = scDblFinder.class == "singlet")
  if (ncol(seurat_singlets[[s]]) == 0) stop("No singlets left in ", s)
}
doublet_calls <- bind_rows(doublet_results)
write.csv(doublet_calls, file.path(out_dir, "scDblFinder_per_cell_calls.csv"), row.names = FALSE)
doublet_summary <- doublet_calls %>%
  count(Patient, scDblFinder_class, name = "n") %>%
  tidyr::pivot_wider(names_from = scDblFinder_class, values_from = n, values_fill = 0)
write.csv(doublet_summary, file.path(out_dir, "scDblFinder_summary.csv"), row.names = FALSE)

# ---- 8. Epithelial VEGFA by patient, one-way ANOVA, Tukey HSD ----
vegfa_epi <- vegfa_by_patient_celltype %>% filter(CellType == "Epithelial cells")

  warning("Not all seven patients have epithelial cells; review annotations before statistical tests")
}
vegfa_epi$Stage <- factor(as.character(vegfa_epi$Stage), levels = c("I", "II", "III"))
anova_result <- aov(Mean_VEGFA ~ Stage, data = vegfa_epi)
anova_p <- summary(anova_result)[[1]][["Pr(>F)"]][1]
tukey_result <- TukeyHSD(anova_result)
write.csv(as.data.frame(summary(anova_result)[[1]]),
          file.path(out_dir, "VEGFA_epithelial_ANOVA.csv"))
write.csv(as.data.frame(tukey_result$Stage),
          file.path(out_dir, "VEGFA_epithelial_TukeyHSD.csv"))

vegfa_summary <- vegfa_epi %>% group_by(Stage) %>% summarise(
  Mean = mean(Mean_VEGFA), SD = sd(Mean_VEGFA), n = n(), .groups = "drop")
y_max <- max(vegfa_summary$Mean + vegfa_summary$SD, vegfa_epi$Mean_VEGFA, na.rm = TRUE)
if (!is.finite(y_max) || y_max <= 0) y_max <- 1
p_label <- if (is.na(anova_p)) "ANOVA P = NA" else if (anova_p < 0.001) {
  "ANOVA P < 0.001"
} else sprintf("ANOVA P = %.3f", anova_p)
